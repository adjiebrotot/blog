{%- comment -%} Sets post_keys: the category keys a post's tags resolve to, each once. A tag not in _data/categories.yml (or no tag at all) counts as "others". Usage: {% include post-keys.html tags=post.tags %} {%- endcomment -%} {%- capture _keys -%} {%- for _tg in include.tags -%} {%- assign _k = _tg | downcase | strip -%} {%- assign _c = site.data.categories | where: "key", _k | first -%} {%- if _c -%}{{ _k }}{%- else -%}others{%- endif -%} {%- unless forloop.last -%},{%- endunless -%} {%- endfor -%} {%- endcapture -%} {%- assign post_keys = _keys | split: "," | uniq -%} {%- if post_keys.size == 0 -%}{%- assign post_keys = "others" | split: "," -%}{%- endif -%}
